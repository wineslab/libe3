24/**
25 * @brief What a PDU is for, so that a link that has streams can pick one
26 *
27 * A connector does not decode the PDUs it carries. The caller says which kind
28 * a PDU is, and a link with more than one stream (SCTP) sends indications on
29 * one and everything else on the other.
30 */
31enum class PduClass : uint8_t {
32 PROCEDURE = 0, ///< A request, a response, a control, a release: anything but an indication
33 INDICATION = 1 ///< An indication message
34};

36/**
37 * @brief Socket and admission options of the POSIX link
38 *
39 * The ZeroMQ link ignores all of them. The defaults are those of E3Config.
40 */
41struct PosixOptions {
42 /// SO_SNDBUF and SO_RCVBUF of SCTP sockets, set before the association exists; 0 keeps the kernel's
43 size_t socket_buffer_bytes{2 * 1024 * 1024};
44 /// SCTP delayed SACK; false turns it off, on both peers, as OCUDU does
45 bool sctp_delayed_sack{false};
46 /// RAN role: literal addresses a peer may connect from (TCP and SCTP); empty accepts any
47 std::vector<std::string> allowed_peers;
48 /// RAN role: most peers connected at once; 0 is unlimited
49 size_t maximum_associations{0};
50};

52/**
53 * @brief Abstract base class for E3AP transport connectors
54 *
55 * This class defines the transport-agnostic interface that all connectors
56 * must implement. The design mirrors the original C implementation's
57 * function pointer approach, but uses virtual methods for type safety.
58 *
59 * Derived classes:
60 * - PosixE3Connector: POSIX socket-based transport (TCP, SCTP, IPC); one
61 * endpoint per dApp carries the setup, inbound and outbound traffic
62 * - ZmqE3Connector: ZeroMQ-based transport
63 */
64class E3Connector {

74 * @brief Move assignment.
75 * @return This connector
76 */
77 E3Connector& operator=(E3Connector&&) = default;

109 virtual ErrorCode setup_inbound_connection() = 0;

111 /**
112 * @brief Receive data from the inbound channel
113 * @param buffer Buffer to store received data
114 * @return Number of bytes received; 0 on timeout (nothing yet);
115 * ErrorCode::NOT_CONNECTED once the peer has closed (dApp role);
116 * any other negative ErrorCode on a transport error
117 */
118 virtual int receive(std::vector<uint8_t>& buffer) = 0;
119
120 /**
121 * @brief Set up the outbound connection (PUB channel)
122 *
123 * This establishes the channel for sending indications to dApps.
124 * @return ErrorCode::SUCCESS on success, error code otherwise
125 */
126 virtual ErrorCode setup_outbound_connection() = 0;
127
128 /**
129 * @brief Send data on the outbound channel
130 * @param data Data to send
131 * @return ErrorCode::SUCCESS on success, error code otherwise
132 */
133 virtual ErrorCode send(const std::vector<uint8_t>& data) = 0;
134
135 /**
136 * @brief Send data to one peer, or to all of them (RAN role)
137 *
138 * The default broadcasts with send(), which is what a connector that cannot
139 * tell its peers apart does.
140 *
141 * @param peer_token The peer named by last_peer_token() or last_setup_peer_token(); 0 sends to every peer
142 * @param data Data to send
143 * @param kind What the PDU is for
144 * @return ErrorCode::SUCCESS on success, an error code otherwise
145 */
146 virtual ErrorCode send_to(uint64_t peer_token, const std::vector<uint8_t>& data, PduClass kind) {
147 (void)peer_token;
148 (void)kind;
149 return send(data);
150 }

153 // Client-side (dApp role) operations
154 //
155 // Mirrors of the server-side methods above but with bind/listen replaced
156 // by connect. Default implementations return INVALID_PARAM so connectors
157 // that have not been taught the dApp role surface a clear error instead
158 // of silently misbehaving. ZmqE3Connector and PosixE3Connector override

212 * Called before joining threads to unblock any blocking socket operations.

217 /**
218 * @brief Which peer supplied the bytes the last receive() returned (RAN role)
219 *
220 * Call it on the thread that calls receive(), right after a receive() that
221 * returned bytes. A token names one accepted connection and is never reused
222 * by the same connector, so a closed peer's token cannot name a later one.
223 *
224 * @return The peer's token, or 0 for a connector that cannot tell its
225 * peers apart (the default, and ZeroMQ)
226 */
227 virtual uint64_t last_peer_token() const { return 0; }
228
229 /**
230 * @brief Which peer sent the setup request the last recv_setup_request() returned (RAN role)
231 *
232 * Call it on the thread that calls recv_setup_request(), right after a call
233 * that returned bytes. The token is the one last_peer_token() gives for the
234 * same peer's later bytes, so a RAN can tie a dApp to its connection at setup.
235 *
236 * @return The peer's token, or 0 for a connector that cannot tell its peers
237 * apart (the default, and ZeroMQ)
238 */
239 virtual uint64_t last_setup_peer_token() const { return 0; }
240
241 /**
242 * @brief Peers whose connection has closed since the last call (RAN role)
243 *
244 * Call it on the thread that calls receive(). Each token is returned once,
245 * and only after the bytes that peer sent before closing were delivered.
246 * The default returns nothing: a connector that cannot tell its peers apart
247 * cannot tell that one left.
248 *
249 * @return The tokens of the peers that closed, oldest first
250 */
251 virtual std::vector<uint64_t> take_closed_peers() { return {}; }

311/**
312 * @brief Factory function to create appropriate connector
313 *
314 * Creates a connector instance based on the link and transport layers specified.
315 *
316 * @param link_layer Link layer to use (ZMQ or POSIX)
317 * @param transport_layer Transport layer to use (SCTP, TCP, or IPC)
318 * @param setup_endpoint Endpoint for setup channel
319 * @param inbound_endpoint Endpoint for inbound channel
320 * @param outbound_endpoint Endpoint for outbound channel
321 * @param setup_port Port of the setup channel (SCTP and TCP; IPC uses the endpoint path)
322 * @param inbound_port Port of the inbound channel (SCTP and TCP)
323 * @param outbound_port Port of the outbound channel (SCTP and TCP)
324 * @param io_threads Number of I/O threads (for ZMQ)
325 * @param role Role of the peer (RAN or dApp)
326 * @param posix Socket and admission options of the POSIX link (ignored by ZMQ)
327 * @return Unique pointer to created connector, nullptr on failure
328 *
329 * The POSIX link serves everything on the setup endpoint and port, and ignores
330 * the inbound and outbound ones.
331 */
332std::unique_ptr<E3Connector> create_connector(
333 E3LinkLayer link_layer,
334 E3TransportLayer transport_layer,
335 const std::string& setup_endpoint,
336 const std::string& inbound_endpoint,
337 const std::string& outbound_endpoint,
338 uint16_t setup_port,
339 uint16_t inbound_port,
340 uint16_t outbound_port,
341 size_t io_threads = 2,
342 E3Role role = E3Role::RAN,
343 const PosixOptions& posix = PosixOptions{}
344);
