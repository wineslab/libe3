29/**
30 * @brief Current realtime clock reading, in nanoseconds since the Unix epoch.
31 *
32 * This is the clock behind every producer-side timestamp libE3 puts on the
33 * wire, chosen so the value stays comparable across hosts (bounded by whatever
34 * NTP or PTP discipline is in place) rather than being meaningful only within
35 * one process.
36 *
37 * @warning This is @c CLOCK_REALTIME, not a monotonic clock. It can step
38 * backwards or jump when the system clock is adjusted, and it must
39 * never be subtracted from a monotonic reading to derive a duration.
40 *
41 * @return Nanoseconds since the Unix epoch, UTC
42 */
43inline uint64_t now_realtime_ns() noexcept {
44 return static_cast<uint64_t>(
45 std::chrono::duration_cast<std::chrono::nanoseconds>(
46 std::chrono::system_clock::now().time_since_epoch()
47 ).count());
48}

401/**
402 * @brief Generic E3AP PDU structure
403 */
404struct Pdu {
405 PduType type{PduType::SETUP_REQUEST}; ///< Which alternative of `choice` is set
406 PduChoice choice; ///< The message itself
407 uint32_t message_id{0}; ///< Unique message identifier
408 uint64_t enqueue_seq{0}; ///< Set by queue_outbound; keys the [latrec] L-stage records
409
410 /**
411 * @brief Producer timestamp, in nanoseconds since the Unix epoch.
412 *
413 * Stamped from the realtime clock (see @ref now_realtime_ns) when the PDU
414 * is constructed, so a caller gets one for free. Zero means "no producer
415 * reference": every encoding transmits it as an absent field and decodes
416 * an absent field back to zero, so the clock read is opt-out and a PDU
417 * without a timestamp is a normal message rather than a malformed one.
418 *
419 * @warning Realtime, not monotonic. Never subtract it from a monotonic
420 * reading.
421 */
422 uint64_t timestamp{0};
423
424 /// SetupRequest-typed PDU with a default-constructed message and a fresh timestamp.
425 Pdu() : timestamp(now_realtime_ns()) {}
426
427 /// PDU of type `t` with a default-constructed message and a fresh timestamp.
428 /// @param t PDU type
429 explicit Pdu(PduType t) : type(t), timestamp(now_realtime_ns()) {}
430
431 /// PDU of type `t` that holds `data`, with a fresh timestamp.
432 /// @param t PDU type; it must match the type of `data`
433 /// @param data The message
434 template<typename T>
435 Pdu(PduType t, T&& data)
436 : type(t), choice(std::forward<T>(data)), timestamp(now_realtime_ns()) {}
437
438 /**
439 * @brief Get PDU data with type checking
440 * @tparam T Message type to look for
441 * @return The message, or nullptr when the PDU holds another type
442 */
443 template<typename T>
444 T* get_if() noexcept {
445 return std::get_if<T>(&choice);
446 }
447
448 /**
449 * @brief Get PDU data with type checking
450 * @tparam T Message type to look for
451 * @return The message, or nullptr when the PDU holds another type
452 */
453 template<typename T>
454 const T* get_if() const noexcept {
455 return std::get_if<T>(&choice);
456 }
457};
