39/// What a DAppSession holds: the agent, the event queue and the setup snapshot.
40struct DAppSession::Impl {
41 /**
42 * @brief Create the agent and the event queue
43 * @param config The agent's configuration
44 * @param queue_capacity Number of events the queue holds
45 */
46 explicit Impl(libe3::E3Config config, std::size_t queue_capacity)
47 : agent(std::make_unique<libe3::E3Agent>(std::move(config))),
48 queue(queue_capacity) {}
49
50 std::unique_ptr<libe3::E3Agent> agent; ///< The dApp agent
51 // Stable object (never moved) so a poll_events() reference stays valid; a
52 // prior stop() shut it down, and start() re-arms it in place (LockFreeQueue's
53 // shutdown is one-way, E3Agent itself is restartable).
54 libe3::LockFreeQueue<E3Event> queue; ///< Events waiting for poll_events()
55 std::atomic<unsigned long long> dropped{0}; ///< Events dropped because the queue was full
56
57 // SetupResponse snapshot for the setup accessors (written once on setup).
58 mutable std::mutex setup_mu; ///< Guards have_setup and setup
59 bool have_setup{false}; ///< Whether a setup response has been seen
60 libe3::SetupResponse setup; ///< The latest setup response
61
62 /**
63 * @brief Queue an event for poll_events()
64 * @param ev The event; it is dropped, and counted, if the queue is full
65 */
66 void enqueue(E3Event ev) {
67 const uint64_t seq = latrec_seq_next();
68 ev.trace_seq = seq;
69 const uint64_t kind = static_cast<uint64_t>(ev.kind);
70 // Before the push, for the same reason as the report queue: the
71 // binding's poll thread can drain this event and stamp
72 // SESSION_POLLED before this thread gets to issue its own stamp.
73 const uint64_t t_queued = latrec_tnow();
74 if (queue.push(std::move(ev)) != libe3::ErrorCode::SUCCESS) {
75 dropped.fetch_add(1, std::memory_order_relaxed);
76 latrec_tstamp(seq, LATREC_DROP, 0, LATREC_DROP_SESSION_QUEUE);
77 } else {
78 latrec_tstamp_at(seq, LATREC_SESSION_QUEUED, kind, 0, t_queued);
79 }
80 }
81};

255int64_t DAppSession::subscribe(uint32_t ran_function_id,
256 std::vector<uint32_t> telemetry_ids,
257 std::vector<uint32_t> control_ids,
258 int sub_time_ms,
259 int periodicity) {
260 uint32_t request_id = 0;
261 ErrorCode rc = impl_->agent->subscribe(
262 ran_function_id, std::move(telemetry_ids), std::move(control_ids),
263 opt_u32(sub_time_ms), opt_u32(periodicity), &request_id);
264 // On success return the assigned request id (positive, 1..4294967295) so Python
265 // can correlate the SubscriptionResponse by id; on failure return the
266 // ErrorCode (all error codes are negative, SUCCESS is 0).
267 if (rc == ErrorCode::SUCCESS) {
268 return static_cast<int64_t>(request_id);
269 }
270 return static_cast<int64_t>(rc);

323std::vector<E3Event> DAppSession::poll_events(std::size_t max_batch, int timeout_ms) {
324 std::vector<E3Event> out;
325 if (max_batch == 0) return out;
326 auto& queue = impl_->queue;
327 // libe3 does not start this thread, so it has no ring. The open happens
328 // once, on a per-batch call rather than a per-event one.
329 latrec_tls_open_as("libe3.session");
330
331 // timeout_ms <= 0 is a non-blocking poll (try_pop); a positive value blocks
332 // up to that long for the first event. (pop() spins/yields before checking
333 // the deadline, so 0 there would busy-wait.)
334 std::optional<E3Event> first =
335 (timeout_ms <= 0) ? queue.try_pop()
336 : queue.pop(std::chrono::milliseconds(timeout_ms));
337 if (!first) return out; // quiet tick, or shutdown after stop()
338
339 out.reserve(std::min(max_batch, queue.capacity()));
340 out.push_back(std::move(*first));
341 while (out.size() < max_batch) {
342 auto next = queue.try_pop();
343 if (!next) break;
344 out.push_back(std::move(*next));
345 }
346 // aux is the position in the batch and aux2 its size, so a backlog shows
347 // as batches reaching max_batch.
348 for (std::size_t i = 0; i < out.size(); ++i) {
349 latrec_tstamp(out[i].trace_seq, LATREC_SESSION_POLLED, i, out.size());
350 }
351 return out;
352}
