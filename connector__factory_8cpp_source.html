17std::unique_ptr<E3Connector> create_connector(
18 E3LinkLayer link_layer,
19 E3TransportLayer transport_layer,
20 const std::string& setup_endpoint,
21 const std::string& inbound_endpoint,
22 const std::string& outbound_endpoint,
23 uint16_t setup_port,
24 uint16_t inbound_port,
25 uint16_t outbound_port,
26 size_t io_threads,
27 E3Role role
28) {
29 std::unique_ptr<E3Connector> conn;
30 switch (link_layer) {
31 case E3LinkLayer::ZMQ:
32#if LIBE3_HAS_ZMQ
33 conn = std::make_unique<ZmqE3Connector>(
34 transport_layer, setup_endpoint, inbound_endpoint, outbound_endpoint,
35 setup_port, inbound_port, outbound_port, io_threads
36 );
37 break;
38#else
39 // zmq_connector.cpp is not compiled in this configuration, so this
40 // is a configuration error at runtime rather than a link error at
41 // build time. E3LinkLayer::ZMQ stays in the enum either way: the
42 // value can arrive from a config file the build knows nothing about.
43 E3_LOG_ERROR("ConnFactory")
44 << "ZMQ link layer requested but libe3 was built with "
45 "LIBE3_ENABLE_ZMQ=OFF";
46 return nullptr;
47#endif
48
49 case E3LinkLayer::POSIX:
50 conn = std::make_unique<PosixE3Connector>(
51 transport_layer, setup_endpoint, inbound_endpoint, outbound_endpoint,
52 setup_port, inbound_port, outbound_port
53 );
54 break;
55
56 default:
57 E3_LOG_ERROR("ConnFactory") << "Unsupported link layer: "
58 << static_cast<int>(link_layer);
59 return nullptr;
60 }
61 if (conn) conn->set_role(role);
62 return conn;
63}
