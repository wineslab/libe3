30/**
31 * @brief Abstract Service Model interface
32 *
33 * This is the extension point for RAN vendors to implement their
34 * Service Models. The E3 agent treats E3SM as opaque - all SM-specific
35 * encoding/decoding happens in the implementations of this interface.
36 *
37 * RAN vendors should:
38 * 1. Derive from this class for each supported SM
39 * 2. Implement the required virtual methods
40 * 3. Register their SMs with the E3Agent
41 */
42class ServiceModel {
43public:
44 virtual ~ServiceModel() = default;

72 * @return The control identifiers the SM supports

74 virtual std::vector<uint32_t> control_ids() const = 0;

76 /**
77 * @brief RAN-function-specific opaque data (required, 1..32768 bytes)
78 *
79 * Returns the byte vector advertised for this RAN function in the
80 * SetupResponse (ranFunctionList->ranFunctionData). E3AP declares the field
81 * mandatory and non-empty -- `OCTET STRING (SIZE (1..262144))`, limited to
82 * 32768 bytes -- so every SM
83 * must override this: even the most minimal SM carries at least its own
84 * name. An SM with nothing to advertise is not advertisable at all, since
85 * ranFunctionList itself is optional but its entries are not.
86 *
87 * Registration fails with ErrorCode::INVALID_PARAM if the returned vector is
88 * empty or longer than MAX_PROTOCOL_DATA_SIZE. The base implementation
89 * returns an empty vector so that failure is a loud diagnostic rather than a
90 * compile error in every existing SM; it is not a usable default.
91 *
92 * @return The bytes advertised for this RAN function; registration fails if
93 * the vector is empty or oversized.
94 */
95 virtual std::vector<uint8_t> ran_function_data() const { return {}; }
96
97 /**
98 * @brief The least time between two indications that this SM keeps for a subscription
99 *
100 * Sent to the dApp as the granted periodicity of its SubscriptionResponse.
101 * Override it only when the SM itself keeps its indications at least that
102 * far apart. The base implementation reports nothing, so the response then
103 * carries no periodicity ("not reported").
104 *
105 * @param requested_us Periodicity the dApp asked for, in microseconds. 0 when
106 * it asked for none or for one indication per event.
107 * @return Microseconds between two indications. std::nullopt or 0 when the SM
108 * keeps no interval or does not say; the response then has none.
109 */
110 virtual std::optional<uint32_t> granted_periodicity(uint32_t requested_us) const {
111 (void)requested_us;
112 return std::nullopt;
113 }

115 /**
116 * @brief Initialize the SM

118 * Called when the SM is first registered.

121 virtual ErrorCode init() = 0;

151 /**
152 * @brief Process a control action from a dApp
153 *
154 * Called by the E3 interface when a DAppControlAction is received for this
155 * service model. Implementations answer it with respond_to_control() and can emit
156 * outbound PDUs (e.g. IndicationMessage) via emit_outbound().
157 * @param request_message_id Message identifier of the DAppControlAction
158 * @param action The control action
159 * @return ErrorCode::SUCCESS if the SM took the action
160 */
161 virtual ErrorCode handle_control_action(
162 uint32_t request_message_id,
163 const DAppControlAction& action
164 ) = 0;

166protected:
167 ServiceModel() = default;

169 /**
170 * @brief Build a MessageAck PDU for a control/request message.
171 *
172 * DEPRECATED TO BE REMOVED. A Service Model answers a control with
173 * respond_to_control(). This helper only builds the PDU.
174 * @param request_message_id Message identifier of the message acknowledged
175 * @param response_code Result to report
176 * @return The PDU
177 */
178 static Pdu make_message_ack_pdu(
179 uint32_t request_message_id,
180 ResponseCode response_code
181 ) {
182 Pdu pdu(PduType::MESSAGE_ACK);
183 MessageAck ack;
184 ack.request_id = request_message_id;
185 ack.response_code = response_code;
186 pdu.choice = ack;
187 return pdu;
188 }
189
190 /**
191 * @brief Answer a dApp control
192 *
193 * Call it once per control, from any thread, when the outcome is known: from
194 * handle_control_action() or later, as an SM that applies the control
195 * asynchronously does. It sends a ControlResponse only when the dApp asked for
196 * one (`request_ack`), so an SM can call it for every control. The RAN function
197 * and the control identifier are filled in from the control. An SM that never
198 * answers sends nothing: the library does not invent an answer for a control
199 * that may still be applied.
200 * @param dapp_id dApp that sent the control (`DAppControlAction::dapp_identifier`)
201 * @param request_message_id Message identifier of the control, as passed to handle_control_action()
202 * @param result Whether the control was applied, why not, and what the SM reports about it
203 * @return ErrorCode::SUCCESS if the answer was queued; ErrorCode::NOT_FOUND if none is owed (the
204 * dApp did not ask, or the control was answered already); ErrorCode::INVALID_PARAM if the
205 * outcome is over MAX_CONTROL_OUTCOME_SIZE; ErrorCode::NOT_INITIALIZED if no agent runs the SM
206 */
207 ErrorCode respond_to_control(
208 uint32_t dapp_id,
209 uint32_t request_message_id,
210 const ControlResult& result
211 ) {
212 if (!control_responder_) {
213 return ErrorCode::NOT_INITIALIZED;
214 }
215 return control_responder_(dapp_id, request_message_id, result);
216 }
217
218 /**
219 * @brief Build an IndicationMessage PDU.
220 *
221 * This helper only builds the PDU. Implementations can emit it with
222 * emit_outbound() when appropriate.
223 * @param dapp_id dApp the indication is for
224 * @param ran_function_id RAN function the indication comes from
225 * @param protocol_data Opaque payload of the SM
226 * @return The PDU
227 */
228 static Pdu make_indication_pdu(
229 uint32_t dapp_id,
230 uint32_t ran_function_id,
231 std::vector<uint8_t> protocol_data
232 ) {
233 Pdu pdu(PduType::INDICATION_MESSAGE);
234 IndicationMessage msg;
235 msg.dapp_identifier = dapp_id;
236 msg.ran_function_identifier = ran_function_id;
237 msg.protocol_data = std::move(protocol_data);
238 pdu.choice = std::move(msg);
239 return pdu;
240 }
241
242 /**
243 * @brief Emit an outbound PDU produced by the SM.
244 * @param pdu The PDU to send
245 * @return ErrorCode::SUCCESS if it was queued, ErrorCode::NOT_INITIALIZED if no emitter is set
246 */
247 ErrorCode emit_outbound(Pdu&& pdu) {
248 if (!outbound_emitter_) {
249 return ErrorCode::NOT_INITIALIZED;
250 }
251 return outbound_emitter_(std::move(pdu));
252 }
253
254 /**
255 * @brief Get currently subscribed dApps for this SM's RAN function.
256 * @return The identifiers of the subscribed dApps, empty when no provider is set
257 */
258 std::vector<uint32_t> get_subscribers() const {
259 if (!subscribers_provider_) {
260 return {};
261 }
262 return subscribers_provider_();
263 }
264
265 /**
266 * @brief Set the outbound PDU emitter callback (used by E3Interface).
267 * @param emitter Callback that queues a PDU
268 */
269 void set_outbound_emitter(std::function<ErrorCode(Pdu&&)> emitter) {
270 outbound_emitter_ = std::move(emitter);
271 }
272
273 /**
274 * @brief Set the control responder callback (used by E3Interface).
275 * @param responder Callback that answers a dApp control
276 */
277 void set_control_responder(std::function<ErrorCode(uint32_t, uint32_t, const ControlResult&)> responder) {
278 control_responder_ = std::move(responder);
279 }
280
281 /**
282 * @brief Set subscriber provider callback (used by E3Interface).
283 * @param provider Callback that lists the subscribed dApps
284 */
285 void set_subscribers_provider(std::function<std::vector<uint32_t>()> provider) {
286 subscribers_provider_ = std::move(provider);
287 }
288
289private:
290 std::function<ErrorCode(Pdu&&)> outbound_emitter_;
291 std::function<ErrorCode(uint32_t, uint32_t, const ControlResult&)> control_responder_;
292 std::function<std::vector<uint32_t>()> subscribers_provider_;
293
294 friend class E3Interface;
295};
