28bool is_json_ws(char c) {
29 return c == ' ' || c == '\t' || c == '\n' || c == '\r';
30}

32size_t skip_ws(const std::string& text, size_t pos) {
33 while (pos < text.size() && is_json_ws(text[pos])) {
34 ++pos;
35 }
36 return pos;
37}

39// Skips a JSON string starting at text[pos] == '"'. Returns the index just
40// past the closing quote, or nullopt on truncation/malformed escapes.
41std::optional<size_t> skip_json_string(const std::string& text, size_t pos) {
42 const size_t n = text.size();
43 if (pos >= n || text[pos] != '"') return std::nullopt;
44 ++pos;
45 while (pos < n) {
46 char c = text[pos];
47 if (c == '\\') {
48 if (pos + 1 >= n) return std::nullopt;
49 if (text[pos + 1] == 'u') {
50 if (pos + 6 > n) return std::nullopt; // \ u X X X X
51 pos += 6;
52 continue;
53 }
54 pos += 2;
55 continue;
56 }
57 if (c == '"') {
58 return pos + 1;
59 }
60 ++pos;
61 }
62 return std::nullopt;
63}
64
65// Skips one JSON value starting at text[pos] (already at its first byte).
66// Returns the index just past the value, or nullopt on anything unexpected
67// or truncated. Objects/arrays are skipped by depth-counting braces/brackets
68// (with nested strings skipped via skip_json_string so a brace/bracket
69// character inside a string is never mistaken for structure); scalars (
70// numbers, true/false/null) are skipped up to the next structural delimiter
71// or whitespace.
72std::optional<size_t> skip_json_value(const std::string& text, size_t pos) {
73 const size_t n = text.size();
74 if (pos >= n) return std::nullopt;
75 char c = text[pos];
76 if (c == '"') {
77 return skip_json_string(text, pos);
78 }
79 if (c == '{' || c == '[') {
80 char open = c;
81 char close = (open == '{') ? '}' : ']';
82 int depth = 1;
83 ++pos;
84 while (pos < n && depth > 0) {
85 char cc = text[pos];
86 if (cc == '"') {
87 auto after = skip_json_string(text, pos);
88 if (!after) return std::nullopt;
89 pos = *after;
90 continue;
91 }
92 if (cc == open) {
93 ++depth;
94 } else if (cc == close) {
95 --depth;
96 }
97 ++pos;
98 }
99 if (depth != 0) return std::nullopt;
100 return pos;
101 }
102 // Bare token: number, true, false, or null.
103 while (pos < n) {
104 char cc = text[pos];
105 if (cc == ',' || cc == '}' || cc == ']' || is_json_ws(cc)) {
106 break;
107 }
108 ++pos;
109 }
110 return pos;
111}

113// Locates the byte span [start, end) of the value bound to `key` as a
114// top-level member of the JSON object beginning at text[obj_start]. `text`
115// is assumed to have already been proven well-formed JSON by the caller
116// (decode()'s prior nlohmann::json::parse succeeded), so this only has to
117// locate an already-guaranteed-to-exist member, not defend against
118// arbitrary malformed input; every access is still bounds-checked and any
119// unexpected byte pattern bails out to nullopt (never guesses), so a bug
120// here can only cost a caller its fast path, never correctness -- callers
121// are expected to fall back to re-serializing the parsed subtree on a miss.
122// Does not unescape keys: a key that itself contains an escape sequence
123// simply won't match and triggers the fallback (no encoder in this file
124// produces escaped key names).
125std::optional<std::pair<size_t, size_t>> find_top_level_member_span(
126 const std::string& text, size_t obj_start, const std::string& key) {
127 const size_t n = text.size();
128 if (obj_start >= n || text[obj_start] != '{') {
129 return std::nullopt;
130 }
131
132 size_t pos = obj_start + 1;
133 while (true) {
134 pos = skip_ws(text, pos);
135 if (pos >= n) return std::nullopt;
136 if (text[pos] == '}') {
137 return std::nullopt; // key not found
138 }
139 if (text[pos] != '"') return std::nullopt;
140
141 size_t key_start = pos + 1;
142 auto after_key = skip_json_string(text, pos);
143 if (!after_key) return std::nullopt;
144 size_t key_end = *after_key - 1; // index of the closing quote
145
146 pos = skip_ws(text, *after_key);
147 if (pos >= n || text[pos] != ':') return std::nullopt;
148 pos = skip_ws(text, pos + 1);
149
150 size_t value_start = pos;
151 auto after_value = skip_json_value(text, pos);
152 if (!after_value) return std::nullopt;
153
154 bool key_matches = (key_end - key_start == key.size()) &&
155 text.compare(key_start, key.size(), key) == 0;
156 if (key_matches) {
157 return std::make_pair(value_start, *after_value);
158 }
159
160 pos = skip_ws(text, *after_value);
161 if (pos < n && text[pos] == ',') {
162 ++pos;
163 continue;
164 }
165 return std::nullopt; // '}' (key not found) or anything unexpected
166 }
167}

203std::string JsonE3Encoder::binary_to_hex(const std::vector<uint8_t>& data) {
204 std::ostringstream ss;
205 for (uint8_t b : data) {
206 ss << std::hex << std::setfill('0') << std::setw(2) << static_cast<int>(b);
207 }
208 return ss.str();
209}

211std::vector<uint8_t> JsonE3Encoder::hex_to_binary(const std::string& hex) {
212 std::vector<uint8_t> result;
213 result.reserve(hex.size() / 2);
214 for (size_t i = 0; i + 1 < hex.size(); i += 2) {
215 result.push_back(static_cast<uint8_t>(
216 std::stoi(hex.substr(i, 2), nullptr, 16)));
217 }
218 return result;
219}

417IndicationMessage JsonE3Encoder::decode_indication_message(const nlohmann::json& j,
418 const std::string& raw_json) const {
419 IndicationMessage msg;
420 msg.dapp_identifier = j.value("dAppIdentifier", 0u);
421 msg.ran_function_identifier = j.value("ranFunctionIdentifier", 0u);
422
423 size_t obj_start = raw_json.find_first_not_of(" \t\r\n");
424 auto span = (obj_start == std::string::npos)
425 ? std::nullopt
426 : find_top_level_member_span(raw_json, obj_start, "protocolData");
427
428 if (span) {
429 // Verbatim splice: byte-identical to what the SM originally emitted,
430 // instead of re-dumping the parsed subtree (which would silently
431 // reorder keys / drop whitespace for non-canonical input).
432 msg.protocol_data.assign(raw_json.begin() + static_cast<long>(span->first),
433 raw_json.begin() + static_cast<long>(span->second));
434 } else {
435 E3_LOG_DEBUG(LOG_TAG) << "protocolData byte-range scan missed; falling back to DOM dump "
436 "(ranFunctionIdentifier=" << msg.ran_function_identifier << ")";
437 std::string dumped = j["protocolData"].dump();
438 msg.protocol_data.assign(dumped.begin(), dumped.end());
439 }
440 return msg;
441}

493EncodeResult<EncodedMessage> JsonE3Encoder::encode(const Pdu& pdu) {
494 try {
495 nlohmann::json root;
496 root["type"] = to_camel_case(pdu_type_to_string(pdu.type));
497 root["id"] = pdu.message_id;
498 root["timestamp"] = pdu.timestamp;
499
500 // Set only for IndicationMessage: the SM's already-serialized payload,
501 // spliced verbatim into the dumped envelope below instead of being
502 // carried through the nlohmann tree (see encode_indication_message()).
503 std::optional<std::string> raw_protocol_data;
504 bool indication_payload_invalid = false;
505
506 // Encode payload fields directly into root (flat format)
507 std::visit([this, &root, &raw_protocol_data, &indication_payload_invalid](auto&& arg) {
508 using T = std::decay_t<decltype(arg)>;
509
510 nlohmann::json fields;
511 if constexpr (std::is_same_v<T, SetupRequest>) {
512 fields = encode_setup_request(arg);
513 }
514 else if constexpr (std::is_same_v<T, SetupResponse>) {
515 fields = encode_setup_response(arg);
516 }
517 else if constexpr (std::is_same_v<T, SubscriptionRequest>) {
518 fields = encode_subscription_request(arg);
519 }
520 else if constexpr (std::is_same_v<T, SubscriptionDelete>) {
521 fields = encode_subscription_delete(arg);
522 }
523 else if constexpr (std::is_same_v<T, SubscriptionResponse>) {
524 fields = encode_subscription_response(arg);
525 }
526 else if constexpr (std::is_same_v<T, IndicationMessage>) {
527 auto encoded_fields = encode_indication_message(arg);
528 if (!encoded_fields) {
529 indication_payload_invalid = true; // already logged with attribution
530 return;
531 }
532 fields = *encoded_fields;
533 raw_protocol_data.emplace(arg.protocol_data.begin(), arg.protocol_data.end());
534 }
535 else if constexpr (std::is_same_v<T, DAppControlAction>) {
536 fields = encode_dapp_control_action(arg);
537 }
538 else if constexpr (std::is_same_v<T, DAppReport>) {
539 fields = encode_dapp_report(arg);
540 }
541 else if constexpr (std::is_same_v<T, XAppControlAction>) {
542 fields = encode_xapp_control_action(arg);
543 }
544 else if constexpr (std::is_same_v<T, ReleaseMessage>) {
545 fields = encode_release_message(arg);
546 }
547 else if constexpr (std::is_same_v<T, MessageAck>) {
548 fields = encode_message_ack(arg);
549 }
550 root.update(fields);
551 }, pdu.choice);
552
553 if (indication_payload_invalid) {
554 return tl::unexpected(ErrorCode::ENCODE_FAILED);
555 }
556
557 std::string json_str = root.dump();
558
559 if (raw_protocol_data) {
560 // root.dump() sorts object keys lexicographically (documented
561 // nlohmann behavior); with protocolData excluded from the tree,
562 // "ranFunctionIdentifier" is exactly where protocolData would
563 // have sorted to ('d' < 'p' < 'r'), so this reproduces the same
564 // key position (and, for well-formed payloads, the same bytes)
565 // as before.
566 size_t pos = json_str.find("\"ranFunctionIdentifier\"");
567 if (pos == std::string::npos) {
568 E3_LOG_ERROR(LOG_TAG) << "Internal error: indication envelope missing "
569 "ranFunctionIdentifier; cannot splice protocolData";
570 return tl::unexpected(ErrorCode::ENCODE_FAILED);
571 }
572 json_str.insert(pos, "\"protocolData\":" + *raw_protocol_data + ",");
573 }
574
575 EncodedMessage msg;
576 msg.buffer.assign(json_str.begin(), json_str.end());
577 msg.format = EncodingFormat::JSON;
578
579 E3_LOG_TRACE(LOG_TAG) << "Encoded " << pdu_type_to_string(pdu.type)
580 << " (" << msg.size() << " bytes)";
581
582 return msg;
583 }
584 catch (const std::exception& e) {
585 E3_LOG_ERROR(LOG_TAG) << "JSON encode error: " << e.what();
586 return tl::unexpected(ErrorCode::ENCODE_FAILED);
587 }
588}

594EncodeResult<Pdu> JsonE3Encoder::decode(const uint8_t* data, size_t size) {
595 if (data == nullptr || size == 0) {
596 E3_LOG_ERROR(LOG_TAG) << "Cannot decode empty data";
597 return tl::unexpected(ErrorCode::DECODE_FAILED);
598 }
599
600 try {
601 std::string json_str(reinterpret_cast<const char*>(data), size);
602
603 E3_LOG_TRACE(LOG_TAG) << "Decoding JSON (" << size << " bytes)";
604
605 nlohmann::json root = nlohmann::json::parse(json_str);
606
607 Pdu pdu;
608
609 // Get PDU type
610 std::string pdu_type_str = root.value("type", "");
611 auto pdu_type = string_to_pdu_type(pdu_type_str);
612 if (!pdu_type) {
613 E3_LOG_ERROR(LOG_TAG) << "Unrecognized PDU type: " << pdu_type_str;
614 return tl::unexpected(ErrorCode::DECODE_FAILED);
615 }
616 pdu.type = *pdu_type;
617 pdu.message_id = root.value("id", 0u);
618 pdu.timestamp = root.value("timestamp", 0ull);
619
620 if (root.contains("data")) {
621 E3_LOG_ERROR(LOG_TAG) << "Nested \"data\" wrapper is not supported; use flat format";
622 return tl::unexpected(ErrorCode::DECODE_FAILED);
623 }
624
625 // Decode based on PDU type
626 switch (pdu.type) {
627 case PduType::SETUP_REQUEST:
628 pdu.choice = decode_setup_request(root);
629 break;
630 case PduType::SETUP_RESPONSE:
631 pdu.choice = decode_setup_response(root);
632 break;
633 case PduType::SUBSCRIPTION_REQUEST:
634 pdu.choice = decode_subscription_request(root);
635 break;
636 case PduType::SUBSCRIPTION_DELETE:
637 pdu.choice = decode_subscription_delete(root);
638 break;
639 case PduType::SUBSCRIPTION_RESPONSE:
640 pdu.choice = decode_subscription_response(root);
641 break;
642 case PduType::INDICATION_MESSAGE:
643 pdu.choice = decode_indication_message(root, json_str);
644 break;
645 case PduType::DAPP_CONTROL_ACTION:
646 pdu.choice = decode_dapp_control_action(root);
647 break;
648 case PduType::DAPP_REPORT:
649 pdu.choice = decode_dapp_report(root);
650 break;
651 case PduType::XAPP_CONTROL_ACTION:
652 pdu.choice = decode_xapp_control_action(root);
653 break;
654 case PduType::RELEASE_MESSAGE:
655 pdu.choice = decode_release_message(root);
656 break;
657 case PduType::MESSAGE_ACK:
658 pdu.choice = decode_message_ack(root);
659 break;
660 default:
661 E3_LOG_ERROR(LOG_TAG) << "Unknown PDU type: " << pdu_type_str;
662 return tl::unexpected(ErrorCode::DECODE_FAILED);
663 }
664
665 E3_LOG_TRACE(LOG_TAG) << "Decoded " << pdu_type_to_string(pdu.type);
666 return pdu;
667 }
668 catch (const nlohmann::json::parse_error& e) {
669 E3_LOG_ERROR(LOG_TAG) << "JSON parse error: " << e.what();
670 return tl::unexpected(ErrorCode::DECODE_FAILED);
671 }
672 catch (const std::exception& e) {
673 E3_LOG_ERROR(LOG_TAG) << "JSON decode error: " << e.what();
674 return tl::unexpected(ErrorCode::DECODE_FAILED);
675 }
676}
