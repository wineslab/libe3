31struct SubscriptionDetails {
32 std::vector<uint32_t> telemetry_ids;
33 std::vector<uint32_t> control_ids;
34 uint32_t periodicity_us{0}; ///< 0 = use SM default
35 uint32_t subscription_time_s{0}; ///< How long the subscription lasts, 0 = no limit

37 std::optional<std::chrono::steady_clock::time_point> expires_at;
38};

43struct ExpiredSubscription {

49/// Source of "now" for subscription expiry. Empty means std::chrono::steady_clock.

52/**
53 * @brief Callback type for SM lifecycle events
54 *
55 * Called when a RAN function gains its first subscriber or loses
56 * its last subscriber. This allows the E3Agent to start/stop
57 * Service Models as needed.
58 *
59 * @param ran_function_id The RAN function ID
60 * @param should_start True if SM should start, false if it should stop
61 */
62using SmLifecycleCallback = std::function<void(uint32_t ran_function_id, bool should_start)>;

64/**
65 * @brief E3 Subscription Manager
66 *
67 * Thread-safe manager for dApp registrations and RAN function subscriptions.
68 * Handles:
69 * - dApp registration from E3 Setup requests
70 * - Subscription management from E3 Subscription requests
71 * - RAN function lifecycle coordination
72 *
73 * This class is ported from the original C e3_subscription_manager
74 * with modern C++ idioms for thread safety and memory management.
75 */

144 /**
145 * @brief Add a subscription between dApp and RAN function
146 *
147 * @param dapp_id dApp identifier
148 * @param ran_function_id RAN function identifier (0-255)
149 * @param telemetry_ids Telemetry IDs the dApp wants
150 * @param control_ids Control IDs the dApp wants
151 * @param periodicity_us Reporting periodicity in microseconds (0 = SM default)
152 * @param subscription_time_s Seconds the subscription lasts (0 = no limit).
153 * When it runs out the subscription is removed as if the dApp had
154 * deleted it, see expire_subscriptions().
155 * @return std::pair containing:
156 * - ErrorCode::SUCCESS on success, or error code on failure
157 * - The assigned subscription ID (valid only if ErrorCode::SUCCESS)
158 * @return ErrorCode::DAPP_NOT_REGISTERED if dApp not registered
159 * @return ErrorCode::SUBSCRIPTION_EXISTS if already subscribed. A subscription
160 * that has run out but not yet been removed does not count: the
161 * request starts a new one under the same subscription ID.
162 */
163 std::pair<ErrorCode, uint32_t> add_subscription(
164 uint32_t dapp_id,
165 uint32_t ran_function_id,
166 const std::vector<uint32_t>& telemetry_ids = {},
167 const std::vector<uint32_t>& control_ids = {},
168 uint32_t periodicity_us = 0,
169 uint32_t subscription_time_s = 0);
170
171 /**
172 * @brief Remove every subscription whose time has run out
173 *
174 * Removal is silent towards the dApp: E3AP has no "subscription expired"
175 * PDU. The SM lifecycle callback fires for a RAN function that loses its
176 * last subscriber. Call this periodically; get_subscribed_dapps() already
177 * leaves out an expired subscription in between.
178 *
179 * @return the subscriptions that were removed
180 */
181 std::vector<ExpiredSubscription> expire_subscriptions();
