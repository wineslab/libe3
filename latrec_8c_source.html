2 * @file latrec.c
3 * @brief Per-thread ring storage and sequence allocator for latrec.
4 *
5 * latrec.h is usable on its own with an explicitly owned latrec_t. This file
6 * backs the convenience API on top of it: one ring per calling thread, opened
7 * on first stamp and named <prefix>.<tid>, plus the process-wide sequence

20/* Capacity of a per-thread ring, as a power of two. The mapping is pre-faulted
21 * at open, so the ring is resident and a stamp costs the same at any size; the
22 * size sets the footprint and how long a run fits before wrapping. It is
23 * file-backed, so that footprint is page cache and disk, not anonymous memory.
24 *
25 * Every thread that stamps in a LIBE3_ENABLE_LATREC build opens a ring, with
26 * no per-run opt-out, so this default is what an ordinary run of an enabled
27 * build costs per thread. It is deliberately modest for that reason: a long
28 * capture raises it per role (see entries_log2() below and docs/latrec.md)
29 * rather than every test and example paying for a capacity only a benchmark
30 * needs. A ring that wrapped says so: rec_count keeps counting past the
31 * capacity. */
32#ifndef LATREC_TLS_ENTRIES_LOG2
33#define LATREC_TLS_ENTRIES_LOG2 18 /* 256 K records = 8 MiB per thread */
34#endif
35#define LATREC_ENTRIES_LOG2_MIN 12
36#define LATREC_ENTRIES_LOG2_MAX 28
37#define LATREC_MAX_RINGS 64
38
39/* Threads refer here until they open a ring of their own. It stays zeroed, so
40 * enabled == 0 and a stamp from a thread that never called latrec_tls_open()
41 * is discarded at the same check that disables tracing globally. */
42static latrec_t disabled_ring;
43
44__thread latrec_t* latrec_tls = &disabled_ring;

46static latrec_t* open_rings[LATREC_MAX_RINGS];
47static int n_open_rings;
48static pthread_mutex_t rings_mu = PTHREAD_MUTEX_INITIALIZER;
49static pthread_once_t atexit_once = PTHREAD_ONCE_INIT;

51/* Threads do not close their rings, so rec_count and the closing clock pair
52 * are refreshed at exit. Records are valid without this (t_ns != 0); it only
53 * fills in the header counts. */
54static void flush_open_rings(void)
55{
56 pthread_mutex_lock(&rings_mu);
57 for (int i = 0; i < n_open_rings; i++) latrec_heartbeat(open_rings[i]);
58 pthread_mutex_unlock(&rings_mu);
59}

61static void install_atexit(void) { atexit(flush_open_rings); }

63/* LATREC_ENTRIES_LOG2 overrides the compiled-in capacity; out-of-range and
64 * unparseable values fall back to it. */
65static unsigned entries_log2_from(const char* name)
66{
67 const char* e = getenv(name);
68 if (e && *e) {
69 char* end = NULL;
70 long v = strtol(e, &end, 10);
71 if (end && !*end && v >= LATREC_ENTRIES_LOG2_MIN && v <= LATREC_ENTRIES_LOG2_MAX)
72 return (unsigned)v;
73 fprintf(stderr, "[latrec] ignoring %s=%s (expected %d..%d)\n",
74 name, e, LATREC_ENTRIES_LOG2_MIN, LATREC_ENTRIES_LOG2_MAX);
75 }
76 return 0;
77}

79/* Ring capacity for a role: LATREC_ENTRIES_LOG2_<ROLE> if set, else
80 * LATREC_ENTRIES_LOG2, else the compiled-in default. The per-role name is the
81 * role uppercased with every non-alphanumeric character replaced by '_', so
82 * role "oai.mac" reads LATREC_ENTRIES_LOG2_OAI_MAC. Roles differ by orders of
83 * magnitude in stamp rate, so sizing them together either wastes memory or
84 * wraps the busiest ring. */
85static unsigned entries_log2(const char* role)
86{
87 if (role && *role) {
88 char name[128];
89 int n = snprintf(name, sizeof(name), "LATREC_ENTRIES_LOG2_%s", role);
90 if (n > 0 && (size_t)n < sizeof(name)) {
91 for (char* c = name + sizeof("LATREC_ENTRIES_LOG2_") - 1; *c; c++)
92 *c = (*c >= 'a' && *c <= 'z') ? (char)(*c - 'a' + 'A')
93 : ((*c >= 'A' && *c <= 'Z') || (*c >= '0' && *c <= '9')) ? *c : '_';
94 const unsigned v = entries_log2_from(name);
95 if (v)
96 return v;
97 }
98 }
99 const unsigned v = entries_log2_from("LATREC_ENTRIES_LOG2");
100 return v ? v : LATREC_TLS_ENTRIES_LOG2;
101}

103/* Process-wide ring directory. Written once at start-up by whoever owns the
104 * process, read by every thread that opens a ring; the mutex is only there to
105 * make that publication well-defined, since ring opens are not on a hot path. */
106static char output_dir[512];

108void latrec_set_output_dir(const char* dir)
109{
110 pthread_mutex_lock(&rings_mu);
111 if (dir && *dir) snprintf(output_dir, sizeof(output_dir), "%s", dir);
112 else output_dir[0] = '\0';
113 pthread_mutex_unlock(&rings_mu);
114}

116latrec_t* latrec_tls_open_as(const char* role)
117{
118 if (latrec_tls != &disabled_ring) return latrec_tls; /* already opened */
119
120 /* Heap, not TLS: flush_open_rings() and the mapping are still live at
121 * process exit, while a thread's TLS is released when it exits. Not
122 * freed, like the mapping. */
123 latrec_t* r = (latrec_t*)calloc(1, sizeof(*r));
124 if (!r) return latrec_tls; /* stays disabled */
125
126 if (!role || !*role) {
127#ifdef __GLIBC__
128 role = program_invocation_short_name;
129#else
130 role = "latrec";
131#endif
132 }
133 char name[64];
134 snprintf(name, sizeof(name), "%s.%ld", role, (long)syscall(SYS_gettid));
135
136 char dir[sizeof(output_dir)];
137 pthread_mutex_lock(&rings_mu);
138 memcpy(dir, output_dir, sizeof(dir));
139 pthread_mutex_unlock(&rings_mu);
140
141 /* A failed open leaves a disabled ring, which every later latrec_stamp()
142 * skips at the enabled check. */
143 latrec_open_in(r, dir, name, entries_log2(role));
144 latrec_tls = r;
145
146 if (r->enabled) {
147 pthread_once(&atexit_once, install_atexit);
148 pthread_mutex_lock(&rings_mu);
149 /* Beyond the cap a ring still records; it gets no exit flush. */
150 if (n_open_rings < LATREC_MAX_RINGS) open_rings[n_open_rings++] = r;
151 pthread_mutex_unlock(&rings_mu);
152 }
153 return latrec_tls;
154}
