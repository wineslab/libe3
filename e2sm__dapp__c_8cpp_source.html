2 * @file e2sm_dapp_c.cpp
3 * @brief C API of the E2SM-DAPP codec, implemented on top of the C++ one.

28// Grammar limits that bound the size of what is copied from caller memory, so
29// that an absurd length is rejected before it is read, not after.
30constexpr size_t k_max_payload = 32768;
31constexpr size_t k_max_name = 150;
32constexpr size_t k_max_oid = 1000;
33constexpr size_t k_max_subs = 256;
34constexpr size_t k_max_rf_per_sub = 64;
35constexpr size_t k_max_report_sty = 2;
36constexpr size_t k_max_ctrl_sty = 1;

38/// An error to report from inside a conversion; caught at the API boundary.
39struct CError {

41};

43[[noreturn]] void fail(e3_error_t code)

45 throw CError{code};

52template <class T>
53T* alloc_array(size_t n)
54{
55 if (n == 0) {
56 return nullptr;
57 }
58 void* p = std::calloc(n, sizeof(T));
59 if (p == nullptr) {
60 fail(E3_SM_ERROR_MEMORY);
61 }
62 return static_cast<T*>(p);
63}

83std::vector<uint8_t> payload_from(const libe3_e2sm_dapp_bytes_t& b)
84{
85 // An empty payload is a value the encoder rejects (SIZE(1..32768)): let it.
86 if (b.len > k_max_payload) {
87 fail(E3_ENCODE_FAILED);
88 }
89 if (b.len != 0 && b.data == nullptr) {
90 fail(E3_INVALID_PARAM);
91 }
92 return b.len == 0 ? std::vector<uint8_t>() : std::vector<uint8_t>(b.data, b.data + b.len);
93}
94
95std::string string_from(const libe3_e2sm_dapp_bytes_t& b, size_t max)
96{
97 if (b.len > max) {
98 fail(E3_ENCODE_FAILED);
99 }
100 if (b.len != 0 && b.data == nullptr) {
101 fail(E3_INVALID_PARAM);
102 }
103 return b.len == 0 ? std::string() : std::string(reinterpret_cast<const char*>(b.data), b.len);
104}

106dapp::dapp_e3_subscription_list_l subs_from(const libe3_e2sm_dapp_sub_list_t& s)
107{
108 if (s.n_items > k_max_subs) {
109 fail(E3_ENCODE_FAILED);
110 }
111 if (s.n_items != 0 && s.items == nullptr) {
112 fail(E3_INVALID_PARAM);
113 }
114 dapp::dapp_e3_subscription_list_l out;
115 out.resize(s.n_items);
116 for (size_t i = 0; i < s.n_items; ++i) {
117 const auto& in = s.items[i];
118 if (in.n_ran_functions > k_max_rf_per_sub) {
119 fail(E3_ENCODE_FAILED);
120 }
121 if (in.n_ran_functions != 0 && in.ran_functions == nullptr) {
122 fail(E3_INVALID_PARAM);
123 }
124 out[i].dapp_id = in.dapp_id;
125 out[i].subscribed_e3_ran_functions.assign(in.ran_functions, in.ran_functions + in.n_ran_functions);
126 }
127 return out;
128}

174void subs_to_c(const dapp::dapp_e3_subscription_list_l& in, libe3_e2sm_dapp_sub_list_t& out)
175{
176 out.items = alloc_array<libe3_e2sm_dapp_sub_item_t>(in.size());
177 out.n_items = 0;
178 for (size_t i = 0; i < in.size(); ++i) {
179 auto& o = out.items[i];
180 o.dapp_id = static_cast<uint32_t>(in[i].dapp_id);
181 o.ran_functions = alloc_array<uint32_t>(in[i].subscribed_e3_ran_functions.size());
182 o.n_ran_functions = in[i].subscribed_e3_ran_functions.size();
183 for (size_t j = 0; j < o.n_ran_functions; ++j) {
184 o.ran_functions[j] = static_cast<uint32_t>(in[i].subscribed_e3_ran_functions[j]);
185 }
186 out.n_items = i + 1;
187 }
188}

267/// Pack `v` into a malloc'ed buffer in `*out`.
268e3_error_t pack_to_c(const std::vector<uint8_t>& buf, libe3_e2sm_dapp_bytes_t* out)
269{
270 set_bytes(*out, buf.data(), buf.size());
271 return E3_SUCCESS;
272}
273
274template <class Cpp, class CIn, class Convert>
275e3_error_t encode_impl(const CIn* in, libe3_e2sm_dapp_bytes_t* out, Convert convert)
276{
277 if (in == nullptr || out == nullptr) {
278 return E3_INVALID_PARAM;
279 }
280 out->data = nullptr;
281 out->len = 0;
282 try {
283 Cpp v;
284 convert(*in, v);
285 std::vector<uint8_t> buf;
286 if (v.pack(buf) != dapp::ASN_SUCCESS) {
287 return E3_ENCODE_FAILED;
288 }
289 return pack_to_c(buf, out);
290 } catch (const CError& e) {
291 release(*out);
292 return e.code;
293 } catch (const std::bad_alloc&) {
294 release(*out);
295 return E3_SM_ERROR_MEMORY;
296 } catch (...) {
297 release(*out);
298 return E3_INTERNAL_ERROR;
299 }
300}
301
302template <class Cpp, class COut, class Convert, class Release>
303e3_error_t decode_impl(const uint8_t* buf, size_t len, COut* out, Convert convert, Release release_out)
304{
305 if (out == nullptr) {
306 return E3_INVALID_PARAM;
307 }
308 std::memset(out, 0, sizeof(*out));
309 if (buf == nullptr) {
310 return E3_INVALID_PARAM;
311 }
312 try {
313 Cpp v;
314 if (v.unpack(buf, len) != dapp::ASN_SUCCESS) {
315 return E3_DECODE_FAILED;
316 }
317 convert(v, *out);
318 return E3_SUCCESS;
319 } catch (const CError& e) {
320 release_out(*out);
321 std::memset(out, 0, sizeof(*out));
322 return e.code;
323 } catch (const std::bad_alloc&) {
324 release_out(*out);
325 std::memset(out, 0, sizeof(*out));
326 return E3_SM_ERROR_MEMORY;
327 } catch (...) {
328 release_out(*out);
329 std::memset(out, 0, sizeof(*out));
330 return E3_INTERNAL_ERROR;
331 }
332}
