1/**
2 * @file latrec.h
3 * @brief Lock-free per-thread latency stamp recorder (header-only, C99).
4 *
5 * One mmap-backed ring per writer thread, single producer, no locks: a stamp
6 * is one CLOCK_MONOTONIC read and four stores, with no syscall, allocation or
7 * formatting. Records are 32 bytes (seq:48 | cpu:8 | stage:8, t_ns, aux,
8 * aux2); seq is assigned once at the producer and joins the stages offline,
9 * aux/aux2 carry the per-stage payloads listed in the stage catalog below.
10 *
11 * CLOCK_MONOTONIC shares an origin across processes, so RAN, dApp and xApp
12 * stamps join without calibration.
13 *
14 * t_ns is stored last, with release ordering: a reader that acquire-loads a
15 * nonzero t_ns also observes that record's sc, aux and aux2. That makes one
16 * record self-consistent; it does not stop the slot being reused, so a reader
17 * of a ring that has wrapped may pair one write's t_ns with the next write's
18 * payload. A live reader must therefore keep up, or detect the wrap from
19 * rec_count and the single descent in t_ns.
20 *
21 * Gated once, at compile time, by LIBE3_ENABLE_LATREC: off by default, so a
22 * normal build carries none of this, not even a disabled-ring branch. With it
23 * on, tracing runs -- there is no second per-run opt-in. See the bottom of
24 * this file, and docs/latrec.md for where the rings are written.
25 *
26 * File layout: a 4 KiB latrec_hdr followed by `entries` 32-byte records. v1
27 * field offsets are frozen and v2 fields appended inside the header pad, so a
28 * v1 reader parses a v2 file. Readers detect a byte-swapped LATREC_MAGIC.
29 * 64-bit hosts only: a 32-bit host can tear the t_ns store.
30 *
31 * Installed as <libe3/latrec.h>, so a component that links libe3 includes it
32 * from there. Components that do not link libe3 mirror the layout instead, and
33 * the stage catalog below carries their ids too, so it stays the one place a
34 * new id is checked for collisions.
35 *
36 * @see \ref latrec_guide "docs/latrec.md" for the full guide: the clock
37 * model, ring naming/sizing, the stage catalog table, and the
38 * capture-to-CSV workflow.
39 * @see \ref path_a_e3_loop "docs/path-a-e3-loop.md" and
40 * \ref path_b_e2_e3_loop "docs/path-b-e2-e3-loop.md" for the loop boxes
41 * the stage catalog names.
42 *
43 * SPDX-License-Identifier: Apache-2.0
44 */

126/* ---- stage catalog ------------------------------------------------------- */
127/*
128 * One identifier per *operation* on the E3AP / E3SM / E2SM-DAPP path -- not
129 * one per (operation, component) pair. Several components implement the same
130 * operation: an indication is encoded by whichever Service Model produced it,
131 * and E3AP framing is the same libe3 call whether the RAN or the dApp made it.
132 * They share the identifier. Which component wrote a given record comes from
133 * its ring name (<role>.<tid>.latrec), and which concrete message it carried
134 * from the aux payloads, so a component prefix on the stage id would only
135 * duplicate what the reader already knows.
136 *
137 * Identifiers name the boxes of the two documented loops:
138 * docs/path-a-e3-loop.md the E3-only loop (RAN <-> dApp), boxes A1..A20
139 * docs/path-b-e2-e3-loop.md the E2-E3 loop (dApp <-> xApp), boxes B1..B15
140 * Each comment below cites the box(es) it bounds. Boxes that hand off inside
141 * one function on one thread share a single boundary stamp rather than
142 * carrying a separate exit and entry.
143 *
144 * Stable IDs: from this baseline on, never renumber, only append.
145 */
146enum {
147 /* --- Source side: recording and processing what a Service Model reports.
148 * A1 (E3 data recording) and A2 (Processing). Stamped by whichever
149 * component is the data source -- an OAI PHY thread, an OCUDU jbpf hook
150 * feeding its E3 controller, a cuBB L1 data-lake reader, or the Simple SM
151 * shipped in examples/sm_simple. seq = the source's own record counter;
152 * aux = a source-defined slot/frame key (e.g. sfn<<16|slot). */
153 LATREC_RECORD_BEGIN = 0x10, /* A1 entry: raw input capture begins */
154 LATREC_PROCESS_BEGIN = 0x11, /* A1 exit / A2 entry: input in hand,
155 SM-side processing begins */
156
157 /* --- E3SM codec. A3 and A12 (encode), A9 and A18 (decode). The same
158 * Service Model codec on either leg and in either role; aux2 = PduType,
159 * which is what separates an indication from a control or a report.
160 * ENCODE_E3SM_BEGIN doubles as A2's exit. */
161 LATREC_ENCODE_E3SM_BEGIN = 0x18, /* A2 exit / A3 (or A12) entry */
162 LATREC_ENCODE_E3SM_DONE = 0x19, /* A3 (or A12) exit; aux = encoded bytes.
163 Hands to EMIT_ENTER */
164 LATREC_DECODE_E3SM_BEGIN = 0x1A, /* A9 (or A18) entry; aux = payload bytes */
165 LATREC_DECODE_E3SM_DONE = 0x1B, /* A9 (or A18) exit; aux = a decoded count
166 the SM cares about (PRBs, slots, ...) */
167
168 /* --- E3AP codec, queue and transport: libe3 itself. A4..A8 on the
169 * forward leg, A13..A17 on the return leg, B10..B13 on the policy leg --
170 * the same five outbound and three inbound stamps each time.
171 * seq = Pdu::enqueue_seq outbound (allocated at EMIT_ENTER so this record
172 * and the rest of the leg share one key), allocated at RECV inbound. */
173 LATREC_EMIT_ENTER = 0x20, /* A4 entry: the producer's thread crosses
174 into libe3. aux = the producer's own
175 trace seq (whatever it last passed to
176 latrec_ctx_set(), 0 if none), which is
177 what joins the producer's stages to
178 this leg; aux2 = RAN function id.
179 EMIT_ENTER->ENQUEUE is Pdu construction */
180 LATREC_ENQUEUE = 0x21, /* A4 first half exit / A5 entry;
181 aux2 = PduType */
182 LATREC_DEQUEUE = 0x22, /* A5 exit / A4 second half entry, on the
183 publisher thread; aux2 = PduType */
184 LATREC_ENCODE_E3AP_DONE = 0x23, /* A4 exit / A6 entry; aux = wire bytes */
185 LATREC_SEND_DONE = 0x24, /* A6 exit: the connector's send() returned;
186 aux = message_id. NOT the wire: a ZMQ PUB
187 send only copies into ZMQ's queue and the
188 socket write happens later on ZMQ's own
189 io thread, so ENCODE_E3AP_DONE->SEND_DONE
190 excludes that hop for ZMQ but includes it
191 for the POSIX connectors */
192 LATREC_RECV = 0x25, /* A7 / A8 entry; aux = wire bytes read */
193 LATREC_DECODE_E3AP_DONE = 0x26, /* A8 exit; aux = message_id,
194 aux2 = PduType */
195 LATREC_DELIVER_BEGIN = 0x27, /* libe3's inbound filter passed; about to
196 invoke the application. A9's entry seen
197 from the library side. A message the
198 dApp-id filter rejects gets neither this
199 nor DELIVER_DONE: see
200 LATREC_DROP_FILTERED */
201 LATREC_DELIVER_DONE = 0x28, /* the application handler returned;
202 aux2 = PduType */
203 LATREC_REPORT_QUEUED = 0x29, /* A19b entry: report handed to the report
204 worker */
205 LATREC_REPORT_DONE = 0x2A, /* A19b exit */
206 LATREC_SESSION_QUEUED = 0x2B, /* the batching seam a language binding
207 drains, inside A9 / B13; aux = kind */
208 LATREC_SESSION_POLLED = 0x2C, /* handed to the binding; aux = position in
209 batch, aux2 = batch size */
210
211 LATREC_DROP = 0x2F, /* a PDU went no further; aux2 = reason
212 (LATREC_DROP_*). An outcome marker, not
213 a box boundary */
214
215 /* --- Bootstrap: the setup and subscription handshakes, and Service Model
216 * lifecycle. These precede the measured loop rather than sitting on it, so
217 * they are bracketed per handshake instead of per box. Either role stamps
218 * the same pair -- the ring tells you which side you are looking at.
219 * seq = the request's message id. */
220 LATREC_SETUP_BEGIN = 0x30, /* setup request sent (initiator) or
221 reached the handler (responder) */
222 LATREC_SETUP_DONE = 0x31, /* setup response written, or the session
223 is ready to carry traffic; aux2 = 1
224 accepted, 0 rejected. A setup that never
225 decoded into a request is rejected with
226 no SETUP_BEGIN to pair with */
227 LATREC_SUB_BEGIN = 0x32, /* subscription request sent or received */
228 LATREC_SUB_DONE = 0x33, /* subscription response queued or
229 confirmed; aux2 = 1 accepted,
230 0 rejected; aux = granted id */
231 LATREC_SM_START = 0x34, /* first subscription: SM worker up */
232 LATREC_SM_STOP = 0x35, /* last unsubscribe: worker joined;
233 aux2 = batches emitted */
234 LATREC_SM_STATUS_BEGIN = 0x36, /* peer connect/disconnect handler entered;
235 aux = RAN function id */
236 LATREC_SM_STATUS_DONE = 0x37, /* handler returned, cadences recomputed
237 and any RIC update sent */
238
239 /* --- Receiving application, dApp side: A10 (Process), A11 (Create control
240 * or report), and B15 (Apply policy). Bespoke application logic rather
241 * than a library call, so these unify across dApp implementations (the
242 * Python framework and the C++ e3_manager dApps) but never across roles.
243 * seq = the key of the message being handled. */
244 LATREC_PROCESS_DONE = 0x40, /* A10 exit: the dApp reached a decision.
245 A10's entry is DECODE_E3SM_DONE */
246 LATREC_CREATE_OUTPUT = 0x41, /* A11: control or report built and queued;
247 aux = bytes. Hands to ENCODE_E3SM_BEGIN */
248 LATREC_APPLY_POLICY_DONE = 0x42, /* B15 exit: an xApp policy applied in the
249 dApp. B15's entry is DECODE_E3SM_DONE */
250 LATREC_ADMITTED = 0x43, /* the handler decided whether to do the
251 work at all; aux2 = outcome
252 (LATREC_OUT_*). An outcome marker */
253
254 /* --- Receiving application, RAN side: A19a (Apply control). The MAC's own
255 * slot-internal scheduler timing is not an E3 operation and is not in this
256 * catalog. seq = the E3 message id of the control being applied. */
257 LATREC_APPLY_CONTROL_DONE = 0x48, /* A19a: mask or policy installed in the
258 MAC; aux = ok. Takes the scheduler
259 lock, so this couples the control
260 thread to the slot-deadline thread */
261 LATREC_LIVE_ON_AIR = 0x49, /* A19a terminal: the first scheduler tick
262 that put this install (or its clear)
263 on air. aux = sfn<<16|slot,
264 aux2 = installs coalesced into this
265 tick (>1 means the ones behind the
266 keyed install rode along). Stamped on
267 the MAC thread but keyed on the control
268 that installed the mask. A clear with
269 no control behind it carries seq 0 */
270
271 /* --- Acknowledgment tail. Either direction: the receiver emits an ack,
272 * the sender observes it. seq = the acknowledged message's id. */
273 LATREC_ACK_SENT = 0x4A, /* aux = response code, aux2 = reject
274 reason, 0 when accepted */
275 LATREC_ACK_RECV = 0x4B,
276
277 /* --- E2-E3 bridge: A20 (a dApp report going up toward E2) and B9 (an xApp
278 * control coming down toward E3). One pair for both directions;
279 * aux2 = 0 report-up, 1 control-down. seq = a bridge-local counter, since
280 * the E2 envelope carries no id the bridge could propagate. */
281 LATREC_BRIDGE_IN = 0x50, /* message reached the bridge; aux = bytes */
282 LATREC_BRIDGE_OUT = 0x51, /* forwarded on; aux = peers or
283 subscriptions it went to */
284
285 /* --- E2AP codec and SCTP transport. B1 and B6 (encode and send), B2 and
286 * B7 (decode). The same codec whether the xApp side or the E2 agent calls
287 * it, on either flow; aux2 = PDU type. seq = the stamping process's own
288 * counter: no id crosses the RIC, so the legs pair by time, not by seq. */
289 LATREC_ENCODE_E2AP_BEGIN = 0x58, /* aux = payload bytes */
290 LATREC_ENCODE_E2AP_DONE = 0x59, /* SCTP send returned; aux = wire bytes */
291 LATREC_DECODE_E2AP_BEGIN = 0x5A, /* aux = wire bytes received */
292 LATREC_DECODE_E2AP_DONE = 0x5B, /* aux2 = PDU type */
293
294 /* --- E2SM (including E2SM-DAPP) codec, and the xApp's own decision. B3
295 * and B8 (decode), B5 (encode), B4 (xApp processing). The codec pairs are
296 * shared by every framework that speaks E2SM -- the E2 agent's own SM
297 * handling and any xApp framework, Python or otherwise. XAPP_PROCESS_DONE
298 * stays separate from the codecs: it is application logic, not a library
299 * call. seq = the request id the stamping side keys on. */
300 LATREC_DECODE_E2SM_BEGIN = 0x60, /* B3 / B8 entry; the E2AP layer hands the
301 inner payload over; aux = bytes */
302 LATREC_DECODE_E2SM_DONE = 0x61, /* B3 / B8 exit; also B4's entry */
303 LATREC_XAPP_PROCESS_DONE = 0x62, /* B4 exit: the xApp returned a decision.
304 Also B5's entry */
305 LATREC_ENCODE_E2SM_BEGIN = 0x63, /* B5 (or B1) entry, where no xApp
306 decision precedes it */
307 LATREC_ENCODE_E2SM_DONE = 0x64, /* B5 / B1 exit; aux = bytes. Hands to
308 ENCODE_E2AP_BEGIN */
309
310 /* --- Off-path records. Not boxes: these answer "how loaded was the
311 * process" and "why did this row end early", so the one-stamp-per-box
312 * rule does not apply to them. */
313 LATREC_WAIT_ENTER = 0x70, /* a worker went back to its wait carrying
314 the item it just handled. WAIT_ENTER ->
315 the next row's RECORD_BEGIN is time
316 blocked; stamped once per item, so a
317 periodic wait that times out and loops
318 does not restamp it */
319 LATREC_SKIPPED = 0x71, /* a fetched item produced no indication;
320 aux2 = reason (LATREC_SKIP_*) */
321 LATREC_CONTEXT = 0x72 /* slow lane, ~1 Hz on its own ring:
322 aux = involuntary context switches
323 since the last tick, aux2 = current CPU
324 frequency in kHz (0 if unavailable).
325 seq = 0: no per-message key */
326};

378typedef struct {
379 uint64_t sc; /* seq:48 (low) | cpu:8 | stage:8 (high byte) */
380 uint64_t t_ns; /* CLOCK_MONOTONIC; released last: nonzero == valid record */
381 uint64_t aux;
382 uint64_t aux2;
383} latrec_rec;

385/* v1 layout through name[]; v2 appends inside the pad. v1 field offsets are
386 * frozen. */
387typedef struct {
388 uint32_t magic, version;
389 uint64_t entries; /* ring capacity (power of two) */
390 uint64_t rec_count; /* final index; refreshed by latrec_heartbeat */
391 uint64_t t0_mono_ns; /* paired clocks at open: mono<->wall mapping */
392 uint64_t t0_real_ns;
393 char name[64];
394 /* --- v2 --- */
395 uint32_t rec_size; /* sizeof(latrec_rec), for reader validation */
396 uint32_t clock_ns_per_call; /* measured timestamp cost (0 = not measured) */
397 uint64_t t1_mono_ns; /* paired clocks refreshed by heartbeat/close */
398 uint64_t t1_real_ns;
399 uint8_t pad[LATREC_HDR_LEN - 4 * 4 - 8 * 6 - 64];
400} latrec_hdr;

418static inline uint64_t latrec_now_ns(void)
419{
420 struct timespec ts;
421 clock_gettime(CLOCK_MONOTONIC, &ts);
422 return (uint64_t)ts.tv_sec * 1000000000ull + (uint64_t)ts.tv_nsec;
423}
424
425static inline uint64_t latrec_real_ns(void)
426{
427 struct timespec ts;
428 clock_gettime(CLOCK_REALTIME, &ts);
429 return (uint64_t)ts.tv_sec * 1000000000ull + (uint64_t)ts.tv_nsec;
430}

432/* Re-read the cpu id into the ring. Call off the hot path: sched_getcpu() is
433 * a syscall on targets where LATREC_CPUID_ON_STAMP is 0. */
434static inline void latrec_refresh_cpu(latrec_t* r)
435{
436#if LATREC_HAVE_GETCPU
437 const int cpu = sched_getcpu();
438 if (cpu >= 0) r->cpu = (uint32_t)cpu;
439#else
440 (void)r;
441#endif
442}

444/* Mean cost of one latrec_now_ns() over 1000 calls, about 30 us in total.
445 * Called once per ring at open and recorded in the header. */
446static inline uint32_t latrec_measure_clock_ns(void)
447{
448 enum { ITERS = 1000 };
449 uint64_t sink = 0;
450 const uint64_t t0 = latrec_now_ns();
451 for (int i = 0; i < ITERS; i++) sink += latrec_now_ns();
452 const uint64_t t1 = latrec_now_ns();
453 if (sink == 0 || t1 <= t0) return 0;
454 return (uint32_t)((t1 - t0) / (uint64_t)ITERS);
455}

457/* Where rings go when the caller does not name a directory. A build-time
458 * setting, like the recorder itself: -DLATREC_DEFAULT_DIR=... at configure
459 * time. Runtime placement is chosen with latrec_open_in(), or process-wide
460 * with latrec_set_output_dir(). */
461#ifndef LATREC_DEFAULT_DIR
462#define LATREC_DEFAULT_DIR "/tmp/latrec"
463#endif
464
465/* Open a ring under `dir`. name ("process.thread") becomes
466 * <dir>/<name>.latrec and must be unique per writer thread across every
467 * process sharing the directory: the file is opened O_TRUNC, so a collision
468 * discards the other writer's ring. entries_log2 sets the capacity (22 = 4 M
469 * records = 128 MiB). `dir` is created if missing (one level only).
470 * Returns 1 on success, -1 on error.
471 *
472 * There is no "tracing disabled" return: in a build with LIBE3_ENABLE_LATREC
473 * the recorder records. Whether it exists at all is the build's decision. */
474static inline int latrec_open_in(latrec_t* r, const char* dir, const char* name,
475 unsigned entries_log2)
476{
477 memset(r, 0, sizeof(*r));
478 if (!dir || !*dir) dir = LATREC_DEFAULT_DIR;
479 mkdir(dir, 0755); /* EEXIST is the normal case; ignore */
480
481 const uint64_t entries = 1ull << entries_log2;
482 const size_t len = LATREC_HDR_LEN + (size_t)entries * sizeof(latrec_rec);
483
484 /* Built under a temporary name and renamed once the header is written:
485 * rename() is atomic, so a <name>.latrec always carries a valid header. */
486 char path[512], tmp[520];
487 snprintf(path, sizeof(path), "%s/%s.latrec", dir, name);
488 snprintf(tmp, sizeof(tmp), "%s.part", path);
489 int fd = open(tmp, O_CREAT | O_RDWR | O_TRUNC, 0644);
490 if (fd < 0) return -1;
491 if (ftruncate(fd, (off_t)len) != 0) { close(fd); unlink(tmp); return -1; }
492 void* map = mmap(NULL, len, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
493 close(fd); /* mapping keeps the file */
494 if (map == MAP_FAILED) { unlink(tmp); return -1; }
495 memset(map, 0, len); /* pre-fault every page */
496
497 const uint32_t clock_ns = latrec_measure_clock_ns();
498
499 latrec_hdr* h = (latrec_hdr*)map;
500 h->magic = LATREC_MAGIC;
501 h->version = LATREC_VERSION;
502 h->entries = entries;
503 h->rec_size = (uint32_t)sizeof(latrec_rec);
504 h->clock_ns_per_call = clock_ns;
505 h->t0_mono_ns = latrec_now_ns();
506 h->t0_real_ns = latrec_real_ns();
507 h->t1_mono_ns = h->t0_mono_ns;
508 h->t1_real_ns = h->t0_real_ns;
509 snprintf(h->name, sizeof(h->name), "%s", name);
510 if (rename(tmp, path) != 0) { munmap(map, len); unlink(tmp); return -1; }
511
512 r->recs = (latrec_rec*)((uint8_t*)map + LATREC_HDR_LEN);
513 r->mask = entries - 1;
514 r->map = map;
515 r->map_len = len;
516 r->enabled = 1;
517 latrec_refresh_cpu(r);
518
519 if (clock_ns > LATREC_CLOCK_SLOW_NS) {
520 fprintf(stderr,
521 "[latrec] WARNING %s: CLOCK_MONOTONIC costs %u ns/call; the vDSO "
522 "is unavailable (check the clocksource)\n", name, clock_ns);
523 }
524 return 1;
525}

533/* Stamp carrying a timestamp the caller already took, for the case where the
534 * moment being recorded is not the moment the stamp can be issued: handing an
535 * item to another thread is only known to have *succeeded* after the handoff,
536 * by which time the consumer may already have run and stamped its own stage.
537 * Reading the clock before the handoff and passing it here keeps the interval
538 * meaningful; issuing a fresh stamp afterwards can order the producer's record
539 * after the consumer's.
540 *
541 * t_ns must be a CLOCK_MONOTONIC reading from latrec_now_ns(). Zero is treated
542 * as "not taken" and replaced with the current time, since zero is also how an
543 * unwritten slot reads. */
544static inline void latrec_stamp_at(latrec_t* r, uint64_t seq, uint8_t stage,
545 uint64_t aux, uint64_t aux2, uint64_t t_ns)
546{
547 if (LATREC_UNLIKELY(!r->enabled)) return;
548 const uint64_t i = r->idx++;
549 latrec_rec* rec = &r->recs[i & r->mask];
550#if LATREC_CPUID_ON_STAMP
551 const uint64_t cpu = (uint64_t)(unsigned)sched_getcpu();
552#else
553 const uint64_t cpu = (uint64_t)r->cpu;
554#endif
555 rec->sc = (seq & 0x0000FFFFFFFFFFFFull)
556 | ((cpu & 0xFFull) << 48)
557 | ((uint64_t)stage << 56);
558 rec->aux = aux;
559 rec->aux2 = aux2;
560 /* Last store, released: makes the record observable as valid only once
561 * sc/aux/aux2 are visible. */
562 LATREC_STORE_RELEASE(&rec->t_ns, t_ns ? t_ns : latrec_now_ns());
563 /* Take the next record's line for write while the caller runs. */
564 LATREC_PREFETCH_W(&r->recs[(i + 1) & r->mask]);
565}
566
567/* Hot path: one clock read and four stores. Single writer per ring.
568 *
569 * The disabled-ring check is repeated here deliberately. latrec_now_ns() is an
570 * argument, so it would be evaluated before the call even on a thread with no
571 * ring, putting a clock read on a path whose whole point is to be free --
572 * tests/bench_latrec.cpp holds that path to a 10 ns ceiling and catches it. */
573static inline void latrec_stamp(latrec_t* r, uint64_t seq, uint8_t stage,
574 uint64_t aux, uint64_t aux2)
575{
576 if (LATREC_UNLIKELY(!r->enabled)) return;
577 latrec_stamp_at(r, seq, stage, aux, aux2, latrec_now_ns());
578}

580/* Refresh rec_count and the closing clock pair without closing the ring. Call
581 * periodically so a ring that is never closed still carries both. */
582static inline void latrec_heartbeat(latrec_t* r)
583{
584 if (!r->enabled) return;
585 latrec_hdr* h = (latrec_hdr*)r->map;
586 h->rec_count = r->idx;
587 h->t1_mono_ns = latrec_now_ns();
588 h->t1_real_ns = latrec_real_ns();
589}

591static inline void latrec_close(latrec_t* r)
592{
593 if (!r->enabled) return;
594 latrec_heartbeat(r);
595 msync(r->map, r->map_len, MS_SYNC);
596 munmap(r->map, r->map_len);
597 r->enabled = 0;
598}
